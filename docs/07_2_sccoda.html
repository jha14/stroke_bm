scCODA / tascCODA¶

组成变化的主模型。Kernel：conda pertpy。先跑 07_1_composition.qmd。scCODA 看大类，tascCODA 看亚型（树是大类→亚型），对照是 CTRL。

每个点仍是一例的占比。模型会处理「占比加起来必须是 100%」，所以比普通 Wilcoxon 更合适。看可信区间是否离开「无变化」。树图往上/往下是相对 CTRL 的增减，先看中性粒、Pre-Neut、GMP。和 07.1 不一致时以本页为准。

# pertpy writes n_total into AnnData.obsm as (n, 1); zeros become 0.5.
if not getattr(CompositionalModel2, "_n_total_squeezed", False):
    _orig_run_mcmc = CompositionalModel2._CompositionalModel2__run_mcmc

    def _run_mcmc_squeeze_n_total(
        self, sample_adata, kernel, rng_key, copy=False, *args, **kwargs
    ):
        _mcmc_run = _CodaMCMC.run

        def _run(
            mcmc_self,
            rng_key,
            counts,
            covariates,
            n_total,
            ref_index,
            sample_adata,
            extra_fields=(),
        ):
            # obsm forces n_total (n,1); 0.5 pseudocounts are not integers.
            # numpyro DirichletMultinomial then rejects total_count.
            counts_np = np.asarray(counts, dtype=np.float64)
            counts_np = np.where(np.isclose(counts_np, 0.5), 1.0, counts_np)
            counts_np = np.rint(counts_np).astype(np.int32)
            n_total = jnp.asarray(counts_np.sum(axis=1).astype(np.int32))
            counts = jnp.asarray(counts_np)
            ref = np.asarray(ref_index).reshape(-1)
            ref_index = int(ref[0]) if ref.size == 1 else jnp.asarray(ref)
            return _mcmc_run(
                mcmc_self,
                rng_key,
                counts,
                covariates,
                n_total,
                ref_index,
                sample_adata,
                extra_fields=extra_fields,
            )

        _CodaMCMC.run = _run
        try:
            return _orig_run_mcmc(
                self, sample_adata, kernel, rng_key, copy, *args, **kwargs
            )
        finally:
            _CodaMCMC.run = _mcmc_run

    CompositionalModel2._CompositionalModel2__run_mcmc = _run_mcmc_squeeze_n_total
    CompositionalModel2._n_total_squeezed = True

def _node_label(n):
    return (n.name or "").removeprefix("annot_main_")


def _node_rank(n):
    name = _node_label(n)
    main = name if name in ANNOT_MAIN_ORDER else SUB_TO_MAIN.get(name)
    main_i = ANNOT_MAIN_ORDER.index(main) if main in ANNOT_MAIN_ORDER else 999
    sub_i = ANNOT_SUB_ORDER.index(name) if name in ANNOT_SUB_ORDER else -1
    return (main_i, sub_i, name)


def sort_coda_tree(tree):
    """Reorder children to match plot_colors.R lineage order (HSPC at top)."""
    for n in tree.traverse("postorder"):
        if n.children:
            n.children.sort(key=_node_rank)
    return tree


def plot_coda_tree(
    tree,
    ax=None,
    node_value=None,
    figsize=(8, 9),
    pos_color="#F07A7A",
    neg_color="#7EB8E8",
):
    """Draw an ete4 tree with matplotlib. Avoids ete4.treeview / Qt (hangs Jupyter)."""
    if ax is None:
        _, ax = plt.subplots(figsize=figsize)
    sort_coda_tree(tree)
    leaves = [n for n in tree.traverse("preorder") if n.is_leaf]
    nleaf = len(leaves)
    ymap = {id(leaf): float(nleaf - 1 - i) for i, leaf in enumerate(leaves)}

    def y_of(node):
        k = id(node)
        if k not in ymap:
            ymap[k] = float(np.mean([y_of(c) for c in node.children]))
        return ymap[k]

    def depth_of(node):
        d = 0
        cur = node
        while cur.up is not None:
            d += 1
            cur = cur.up
        return d

    y_of(tree)
    xs = {id(n): depth_of(n) for n in tree.traverse()}
    vmax = 1.0
    if node_value is not None and len(node_value):
        vmax = max(float(np.nanmax(np.abs(np.asarray(node_value, dtype=float)))), 1e-8)

    for n in tree.traverse():
        xn, yn = xs[id(n)], y_of(n)
        for c in n.children:
            xc, yc = xs[id(c)], y_of(c)
            ax.plot([xn, xn], [yn, yc], color="#8A8A8A", lw=0.8)
            ax.plot([xn, xc], [yc, yc], color="#8A8A8A", lw=0.8)
        name = _node_label(n)
        if n.is_leaf:
            ax.text(xn + 0.08, yn, name, va="center", ha="left", fontsize=8)
        elif name:
            ax.text(
                xn - 0.06, yn + 0.15, name,
                va="bottom", ha="right", fontsize=7, color="#666666",
            )
        if node_value is not None and n.name in getattr(node_value, "index", []):
            v = float(node_value.loc[n.name])
            if v != 0:
                ax.scatter(
                    [xn], [yn],
                    s=25 + 200 * abs(v) / vmax,
                    c=pos_color if v > 0 else neg_color,
                    zorder=3, edgecolors="white", linewidths=0.4,
                )
    ax.set_axis_off()
    ax.set_xlim(-0.8, max(xs.values()) + 2.8)
    ax.set_ylim(-1, nleaf)
    return ax



GROUP_COL = {"CTRL": "#7EB8E8", "ACI": "#F07A7A"}
GROUP_ORDER = ["CTRL", "ACI"]


def _canon_cell(name):
    s = str(name)
    for prefix in ("annot_main_", "annot_sub_"):
        if s.startswith(prefix):
            return s[len(prefix) :]
    return s


def ordered_cell_types(names, kind="sub"):
    """Order cell types as names() of annot_main_col / annot_sub_col."""
    order = ANNOT_MAIN_ORDER if kind == "main" else ANNOT_SUB_ORDER
    rank = {n: i for i, n in enumerate(order)}
    uniq = list(dict.fromkeys(map(str, names)))
    return sorted(uniq, key=lambda n: (rank.get(_canon_cell(n), 999), n))


def reorder_coda_modality(mdata, kind="main", modality_key="coda"):
    """Put coda columns in plot_colors.R order (also fixes pertpy.plot_boxplots)."""
    coda = mdata[modality_key]
    order = ordered_cell_types(list(coda.var_names), kind=kind)
    mdata.mod[modality_key] = coda[:, order].copy()
    return order


def _wilcox_stars(p):
    if p < 0.001:
        return "***"
    if p < 0.01:
        return "**"
    if p < 0.05:
        return "*"
    return "ns"


def plot_coda_boxplots(
    data,
    modality_key="coda",
    cell_kind="sub",
    figsize=(18, 6),
    add_stats=True,
):
    """Grouped boxplots in plot_colors.R order, with Wilcoxon CTRL vs ACI."""
    coda = data[modality_key]
    x = np.asarray(coda.X, dtype=float)
    props = x / np.clip(x.sum(axis=1, keepdims=True), 1e-12, None)
    df = pd.DataFrame(props, columns=[str(c) for c in coda.var_names], index=coda.obs_names)
    df["group"] = pd.Categorical(
        coda.obs["group"].astype(str), categories=GROUP_ORDER, ordered=True
    )
    plot_df = df.melt(id_vars="group", var_name="Cell type", value_name="Proportion")
    order = ordered_cell_types(plot_df["Cell type"].unique(), kind=cell_kind)
    plot_df["Cell type"] = pd.Categorical(
        plot_df["Cell type"], categories=order, ordered=True
    )

    fig, ax = plt.subplots(figsize=figsize)
    sns.boxplot(
        data=plot_df,
        x="Cell type",
        y="Proportion",
        hue="group",
        order=order,
        hue_order=GROUP_ORDER,
        palette=GROUP_COL,
        fliersize=0,
        width=0.7,
        ax=ax,
    )
    sns.stripplot(
        data=plot_df,
        x="Cell type",
        y="Proportion",
        hue="group",
        order=order,
        hue_order=GROUP_ORDER,
        palette=["#444444", "#444444"],
        dodge=True,
        size=3.5,
        alpha=0.85,
        legend=False,
        ax=ax,
    )

    stats_rows = []
    n_hue = 2
    box_width = 0.7
    step = box_width / n_hue
    offsets = [-box_width / 2 + step / 2 + j * step for j in range(n_hue)]
    yspan = max(float(plot_df["Proportion"].max()), 1e-6)
    for i, ct in enumerate(order):
        ctrl = plot_df.loc[
            (plot_df["Cell type"] == ct) & (plot_df["group"] == "CTRL"), "Proportion"
        ]
        aci = plot_df.loc[
            (plot_df["Cell type"] == ct) & (plot_df["group"] == "ACI"), "Proportion"
        ]
        p = float(
            mannwhitneyu(
                aci.to_numpy(), ctrl.to_numpy(), alternative="two-sided"
            ).pvalue
        )
        stats_rows.append(
            {
                "cell": _canon_cell(ct),
                "n_ctrl": int(ctrl.size),
                "n_aci": int(aci.size),
                "median_ctrl": float(ctrl.median()) if len(ctrl) else np.nan,
                "median_aci": float(aci.median()) if len(aci) else np.nan,
                "p": p,
            }
        )
        if not add_stats or p >= 0.05:
            continue
        star = _wilcox_stars(p)
        y = max(ctrl.max(), aci.max()) + 0.04 * yspan
        x0, x1 = i + offsets[0], i + offsets[1]
        h = 0.015 * yspan
        ax.plot([x0, x0, x1, x1], [y, y + h, y + h, y], color="0.35", lw=0.8)
        ax.text(
            (x0 + x1) / 2, y + h, star,
            ha="center", va="bottom", fontsize=8, color="0.15",
        )
    stats = pd.DataFrame(stats_rows)
    stats["p_adj"] = multipletests(stats["p"], method="fdr_bh")[1]

    handles, labels = ax.get_legend_handles_labels()
    ax.legend(
        handles[:2], labels[:2], title="group",
        loc="upper left", bbox_to_anchor=(1, 1),
    )
    ax.set_xlabel(None)
    ax.tick_params(axis="x", rotation=90)
    ax.set_ylim(0, plot_df["Proportion"].max() * 1.28)
    ax.set_title(
        "Wilcoxon rank-sum CTRL vs ACI; stars: uncorrected p "
        "(* <0.05, ** <0.01, *** <0.001). BH FDR in the CSV.",
        loc="left", fontsize=9, pad=8,
    )
    fig.tight_layout()
    return ax, stats
